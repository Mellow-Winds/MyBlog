# 1.struct的自引用
形如
```c
struct SELF {
int a;
struct SELF b;
}
```
和
```c
typedef struct {
int a;
struct SELF b;
}SELF
```
的表述是不合法的，因为struct并未定义完成，将会反复套娃，编译器无法得知struct的大小
但是形如
```c
struct SELF {
int a;
struct SELF *b;
}
```
是可以的
# 2.链表

链表是内存中非连续的线性结构，由节点构成，包括数据和指针
## 👻代码定义：最简单的单向链表
```c
struct Node{
int data;
struct Node* next;
}
```
## 🎈代码定义：双向链表
```c
struct Node{
struct Node* prev;
int data;
struct Node* next;
}
```
## 🤓偷懒写法：
```c
typedef struct unit{
    xxxxxx
    struct unit *next;
} unit;
```
解释一下：
### 1. 上面的 `unit`：这是“户口本上的大名” 📘
- **位置：** `struct unit { ...`
- **含义：** 这是我们在**造**这个东西时给它起的**正式名称**。
- **全称：** 它的完整身份叫 `struct unit`。
- **为什么要写它？**
    - 因为在花括号 `{ ... }` 里面，我们正在定义它。
    - 特别是那行 `struct unit *next;` —— 当时这个结构体还没定义完，“昵称”还没生效呢！
    - 所以，在这个“设计图”**内部**，想要提到自己，必须用全名（大名）`struct unit`。
### 2. 下面的 `unit`：这是“方便好记的小名” 🏷️
- **位置：** `} unit;`
- **含义：** 这是 `typedef`（起名工具）赋予它的**新名字**。
- **全称：** 就叫 `unit`。
- **为什么要写它？**
    - 为了**偷懒**（程序员的美德）！
    - 如果没有这一行，以后每次用它，我们都得写 `struct unit a;`（太累了）。
    - 有了这一行，电脑就知道了：“哦，以后你说 `unit`，我就知道你是说那个复杂的 `struct unit`。”

 上面那个是**给程序自己内部用的**（因为当时小名还没起好），下面那个是**以后给我们用的**（为了少打字）
## 👻示例：循环链表解决约瑟夫环问题
约瑟夫问题：
假设有 $N$ 个人围成一个圈，从第一个人开始报数，数到 $M$ 的那个人出列，然后从他身后的下一个人重新开始报数，直到所有人都出列。问题是：**求出最后的幸存者**

```c
#include<stdio.h>
#include<stdlib.h>
#include<assert.h>
struct node{
    int data;
    struct node * next;
};
struct linkedlist
{
    struct node *head;
    struct node *tail;
};
void initList(struct linkedlist *ll){
    ll->head = NULL;
    ll->tail = NULL;
}
void append(struct linkedlist *ll, struct node *newnode){
    if(!ll->head && !ll->tail){
        ll->head = newnode;
        ll->tail = newnode;
        newnode->next = ll->head;
    }
    else{
        ll->tail->next = newnode;
        ll->tail = newnode;
        newnode->next = ll->head;
    }
}
struct node* initLNode(int value){
    struct node *p = malloc(sizeof(*p));
    p->next = NULL;
    p->data = value;
    return p;
}
void display (struct linkedlist *ll){
    struct node *p = ll->head;
    while(p->next!=ll->head){
        printf("%d ", p->data);
        p = p->next;
    }
    printf("%d ", p->data);
    printf("\n");
}
void sitCricle(struct linkedlist *ll, int num){
    for(int i = 1; i<=num; i++){
        append(ll, initLNode(i));
    }
    //display(ll);
}
void killUntilOne(struct linkedlist *ll, int kill){
    struct node *pnode = ll->head;
    struct node *p_prev;
    assert(ll->tail->next == ll->head);
    while (pnode->next != pnode)//buggy
    {   
        int cnt = 1;
        do{
            cnt++;
            p_prev = pnode;
            pnode = pnode->next;
        }while (cnt < kill);
        p_prev->next = pnode->next;
        //printf("kill %d\n", pnode->data);
        free(pnode);
        pnode = p_prev->next;
    }
    ll->head = pnode; 
    ll->tail = pnode;
    //printf("----\n");
}
int getSurvivor(struct linkedlist *list){
    return list->head->data;
}
void freeList(struct linkedlist *list){
    struct node *p = list->head;
    while(p!=NULL){
        //printf("free%d ", p->data);
        list->head = p->next;
        list->tail->next = list->head;
        if(p == list->tail)
            list->tail = NULL;
        free(p);
        p = list->head;
    }
    printf("\n");
}
int main(){
    int all, kill;
    scanf("%d %d", &all, &kill);
    struct linkedlist list;
    initList(&list);
    sitCricle(&list, all);
    killUntilOne(&list, kill);
    printf("survivor: %d\n", getSurvivor(&list));
    free(list.head);
    //freeList(&list);
}
```
