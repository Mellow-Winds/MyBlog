```python
def target(nums, targetSum):
    seen = []
    numsPair = []
    for i in range(len(nums)):
        for j in range(i + 1, len(nums)):
            if nums[i] + nums[j] == targetSum:
                if nums[i] not in seen and nums[j] not in seen:
                    seen.append(nums[i])
                    seen.append(nums[j])
                    numsPair.append([nums[i], nums[j]])
    return numsPair
```

- `JAVA`代码 用动态规划判断数组中是否有子集的和为给定值，将其改写成递归

```java
public class Solve {
    // 用动态规划判断数组中是否有子集的和为给定值
    public boolean isSubsetEqualsToTarget(int[] nums, int target) {
        int n = nums.length;
        boolean[][] dp = new boolean[n + 1][target + 1];
        for (int i = 0; i <= n; i++) {
            dp[i][0] = true;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= target; j++) {
                if (j < nums[i - 1]) {
                    dp[i][j] = dp[i - 1][j];
                } else {
                    dp[i][j] = dp[i - 1][j] || dp[i - 1][j - nums[i - 1]];
                }
            }
        }
        return dp[n][target];
    }

    public static void main(String[] args) {
        Solve solve = new Solve();
        int[] nums = { 3, 34, 4, 12, 5, 2 };
        int target = 9;
        System.out.println(solve.isSubsetEqualsToTarget(nums, target));
    }
}
```
